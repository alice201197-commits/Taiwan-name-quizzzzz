/*

*/

let submittedNames = [];


/*

總答案數量

/*
====================================================
建立年代卡片
====================================================
*/

function createDecadeCards() {

    const results =
        document.getElementById("results");

    results.innerHTML = "";


    Object.keys(quizData).forEach(decade => {

        const card =
            document.createElement("div");

        card.className =
            "decade-card";

        card.dataset.decade =
            decade;


        /*
        -------------------------------
        男生
        -------------------------------
        */

        let maleSlots = "";

        quizData[decade].male.forEach(
            (name, index) => {

                maleSlots += `

                    <div
                        class="slot male-slot"
                        data-gender="male"
                        data-index="${index}">

                        ${index + 1}

                    </div>

                `;

            }
        );


        /*
        -------------------------------
        女生
        -------------------------------
        */

        let femaleSlots = "";

        quizData[decade].female.forEach(
            (name, index) => {

                femaleSlots += `

                    <div
                        class="slot female-slot"
                        data-gender="female"
                        data-index="${index}">

                        ${index + 1}

                    </div>

                `;

            }
        );


        card.innerHTML = `

            <div class="decade-title">
                民國 ${decade} 年
            </div>


            <div class="gender-section">

                <div class="gender-label">
                    👦 男生
                </div>

                <div class="slots male-slots">
                    ${maleSlots}
                </div>

            </div>


            <div class="gender-section">

                <div class="gender-label">
                    👧 女生
                </div>

                <div class="slots female-slots">
                    ${femaleSlots}
                </div>

            </div>

        `;


        results.appendChild(card);

    });

}


/*
====================================================
提交名字
====================================================
*/

function submitName() {

    const input =
        document.getElementById("nameInput");

    const name =
        input.value.trim();


    /*
    沒有輸入
    */

    if (!name) {

        alert("請先輸入名字！");

        return;

    }


    /*
    如果已經輸入過

    就不重複計算

    */

    if (
        submittedNames.includes(name)
    ) {

        alert("這個名字已經提交過了！");

        input.value = "";

        return;

    }


    /*
    加入已提交名單
    */

    submittedNames.push(name);


    /*
    搜尋所有年代
    */

    let found = false;


    Object.keys(quizData).forEach(
        decade => {

            /*
            搜尋男生
            */

            quizData[decade].male.forEach(
                (answer, index) => {

                    if (
                        answer === name
                    ) {

                        lightUpSlot(
                            decade,
                            "male",
                            index
                        );

                        found = true;

                    }

                }
            );


            /*
            搜尋女生
            */

            quizData[decade].female.forEach(
                (answer, index) => {

                    if (
                        answer === name
                    ) {

                        lightUpSlot(
                            decade,
                            "female",
                            index
                        );

                        found = true;

                    }

                }
            );

        }
    );


    /*
    更新完成率
    */

    updateProgress();


    /*
    更新歷史紀錄
    */

    updateHistory();


    /*
    清空輸入框
    */

    input.value = "";

    input.focus();


    /*
    顯示結果

    如果完全沒有找到

    */

    if (!found) {

        alert(
            `「${name}」不在目前的菜市場名資料中。`
        );

    }

}


/*
====================================================
讓正確答案亮起
====================================================
*/

function lightUpSlot(
    decade,
    gender,
    index
) {

    const card =
        document.querySelector(
            `.decade-card[data-decade="${decade}"]`
        );


    if (!card) {
        return;
    }


    const slots =
        card.querySelectorAll(
            `.slot[data-gender="${gender}"]`
        );


    const slot =
        slots[index];


    if (!slot) {
        return;
    }


    /*
    男生 → 水藍色

    女生 → 淡粉色
    */

    if (gender === "male") {

        slot.classList.add(
            "male-correct"
        );

    } else {

        slot.classList.add(
            "female-correct"
        );

    }


    /*
    顯示名字

    */

    slot.textContent =
        quizData[decade][gender][index];

}


/*
====================================================
計算完成率
====================================================
*/

function updateProgress() {

    let correctCount = 0;


    /*
    計算目前亮起幾格

    */

    document
        .querySelectorAll(
            ".slot.male-correct, .slot.female-correct"
        )
        .forEach(() => {

            correctCount++;

        });


    const percentage =
        Math.round(
            correctCount /
            totalAnswers *
            100
        );


    document
        .getElementById("progressNumber")
        .textContent =
        `${percentage}%`;


    document
        .getElementById("progressFill")
        .style.width =
        `${percentage}%`;

}


/*
====================================================
顯示已輸入名字
====================================================
*/

function updateHistory() {

    const box =
        document.getElementById(
            "answerHistory"
        );

    const list =
        document.getElementById(
            "historyList"
        );


    box.style.display =
        "block";


    list.innerHTML = "";


    submittedNames.forEach(
        name => {

            const item =
                document.createElement(
                    "div"
                );

            item.className =
                "history-name";

            item.textContent =
                name;

            list.appendChild(item);

        }
    );

}


/*
====================================================
交卷
====================================================
*/

function finishQuiz() {

    const finalResult =
        document.getElementById(
            "finalResult"
        );

    const finalAnswers =
        document.getElementById(
            "finalAnswers"
        );


    finalAnswers.innerHTML = "";


    /*
    計算完成率
    */

    let correctCount = 0;


    document
        .querySelectorAll(
            ".slot.male-correct, .slot.female-correct"
        )
        .forEach(() => {

            correctCount++;

        });


    const percentage =
        Math.round(
            correctCount /
            totalAnswers *
            100
        );


    /*
    顯示所有年代答案
    */

    Object.keys(quizData).forEach(
        decade => {

            const section =
                document.createElement(
                    "div"
                );

            section.className =
                "final-decade";


            const maleAnswers =
                quizData[decade].male
                .join("、");


            const femaleAnswers =
                quizData[decade].female
                .join("、");


            section.innerHTML = `

                <div class="final-decade-title">

                    民國 ${decade} 年

                </div>


                <div class="final-gender final-male">

                    👦 男生：
                    ${maleAnswers}

                </div>


                <div class="final-gender final-female">

                    👧 女生：
                    ${femaleAnswers}

                </div>

            `;


            finalAnswers.appendChild(
                section
            );

        }
    );


    /*
    最終完成率

    */

    const score =
        document.createElement("div");

    score.style.textAlign =
        "center";

    score.style.fontSize =
        "20px";

    score.style.fontWeight =
        "bold";

    score.style.marginBottom =
        "25px";

    score.textContent =
        `你的最終完成率：${percentage}%`;


    finalAnswers.prepend(score);


    /*
    顯示

    */

    finalResult.style.display =
        "block";


    finalResult.scrollIntoView({

        behavior: "smooth",

        block: "start"

    });

}


/*
====================================================
重新開始
====================================================
*/

function resetQuiz() {

    const confirmReset =
        confirm(
            "確定要重新開始嗎？目前的答案會全部清除。"
        );


    if (!confirmReset) {
        return;
    }


    submittedNames = [];


    /*
    清除所有亮起的格子

    */

    document
        .querySelectorAll(
            ".slot"
        )
        .forEach(slot => {

            slot.classList.remove(
                "male-correct",
                "female-correct"
            );

        });


    /*
    恢復格子編號

    */

    Object.keys(quizData).forEach(
        decade => {

            const card =
                document.querySelector(
                    `.decade-card[data-decade="${decade}"]`
                );


            card
                .querySelectorAll(
                    ".male-slot"
                )
                .forEach(
                    (slot, index) => {

                        slot.textContent =
                            index + 1;

                    }
                );


            card
                .querySelectorAll(
                    ".female-slot"
                )
                .forEach(
                    (slot, index) => {

                        slot.textContent =
                            index + 1;

                    }
                );

        }
    );


    /*
    清除歷史紀錄

    */

    document
        .getElementById(
            "answerHistory"
        )
        .style.display =
        "none";


    document
        .getElementById(
            "historyList"
        )
        .innerHTML = "";


    /*
    清除最終答案

    */

    document
        .getElementById(
            "finalResult"
        )
        .style.display =
        "none";


    document
        .getElementById(
            "finalAnswers"
        )
        .innerHTML = "";


    /*
    清除輸入框

    */

    document
        .getElementById(
            "nameInput"
        )
        .value = "";


    /*
    更新完成率

    */

    updateProgress();

}
